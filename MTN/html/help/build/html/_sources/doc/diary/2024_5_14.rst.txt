     Public Function 获取菜单列表() As DataTable
        Dim menuDataTable As New DataTable()
        menuDataTable.Columns.Add("Indent", GetType(String)) ' 用于展示层级
        menuDataTable.Columns.Add("id", GetType(Long))
        menuDataTable.Columns.Add("classify", GetType(String))
        menuDataTable.Columns.Add("daohang_name", GetType(String))
        menuDataTable.Columns.Add("daohang_ico", GetType(String))
        menuDataTable.Columns.Add("daohang_bg", GetType(String))
        menuDataTable.Columns.Add("daohang_img", GetType(String))
        menuDataTable.Columns.Add("other_img", GetType(String))
        menuDataTable.Columns.Add("game_emu_id", GetType(Long))
        menuDataTable.Columns.Add("emu_core_id", GetType(Long))
        menuDataTable.Columns.Add("emu_args", GetType(String))
        menuDataTable.Columns.Add("ui_name_id", GetType(Long))
        menuDataTable.Columns.Add("classify_level", GetType(Long))
        menuDataTable.Columns.Add("classify_id", GetType(Long))
        Dim topLevelQuery As String = "SELECT * FROM daohang WHERE classify_level=0 AND classify_del<>'1' ORDER BY id ASC"
        Using topLevelCmd As New SQLiteCommand(topLevelQuery, 数据库对象)
            Using reader As SQLiteDataReader = topLevelCmd.ExecuteReader()
                While reader.Read()
                    Dim row As DataRow = menuDataTable.NewRow()
                    row("Indent") = "" ' 顶级菜单没有缩进
                    row("id") = reader("id")
                    row("classify") = reader("classify")
                    row("daohang_name") = reader("daohang_name")
                    row("daohang_ico") = reader("daohang_ico")
                    row("daohang_bg") = reader("daohang_bg")
                    row("daohang_img") = reader("daohang_img")
                    row("other_img") = reader("other_img")
                    row("game_emu_id") = reader("game_emu_id")
                    row("emu_core_id") = reader("emu_core_id")
                    row("emu_args") = reader("emu_args")
                    row("ui_name_id") = reader("ui_name_id")
                    row("classify_level") = reader("classify_level")
                    row("classify_id") = reader("classify_id")
                    menuDataTable.Rows.Add(row)
                    PopulateSubMenus(menuDataTable, CLng(reader("id")), "")
                End While
            End Using
        End Using
        Return menuDataTable
    End Function
    Private Sub PopulateSubMenus(ByVal dataTable As DataTable, ByVal parentId As Long, ByVal indent As String)
        Dim subMenuQuery As String = "SELECT * FROM daohang WHERE classify_id=@ParentId AND classify_del<>'1' ORDER BY id ASC"
        Using cmd As New SQLiteCommand(subMenuQuery, 数据库对象)
            cmd.Parameters.AddWithValue("@ParentId", parentId)
            Using reader As SQLiteDataReader = cmd.ExecuteReader()
                While reader.Read()
                    Dim row As DataRow = dataTable.NewRow()
                    row("Indent") = indent ' 用于表示层级的缩进
                    row("id") = reader("id")
                    row("classify") = reader("classify")
                    row("daohang_name") = reader("daohang_name")
                    row("daohang_ico") = reader("daohang_ico")
                    row("daohang_bg") = reader("daohang_bg")
                    row("daohang_img") = reader("daohang_img")
                    row("other_img") = reader("other_img")
                    row("game_emu_id") = reader("game_emu_id")
                    row("emu_core_id") = reader("emu_core_id")
                    row("emu_args") = reader("emu_args")
                    row("ui_name_id") = reader("ui_name_id")
                    row("classify_level") = reader("classify_level")
                    row("classify_id") = reader("classify_id")
                    dataTable.Rows.Add(row)

                    ' 递归调用，处理下一级子菜单
                    PopulateSubMenus(dataTable, CLng(reader("id")), indent & "    ")
                End While
            End Using
        End Using
    End Sub
